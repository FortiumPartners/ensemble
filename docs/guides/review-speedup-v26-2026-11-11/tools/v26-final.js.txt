function fisher(a, b, c, d) { // 2x2 [[a,b],[c,d]], two-sided (sum of probabilities not above the observed), exact
  const n = a + b + c + d; const r1 = a + b; const c1 = a + c;
  const p = (x) => Math.exp(logC(r1, x) + logC(n - r1, c1 - x) - logC(n, c1));
  const obs = p(a); let tot = 0; let one = 0;
  for (let x = Math.max(0, c1 - (n - r1)); x <= Math.min(r1, c1); x += 1) { const px = p(x); if (px <= obs + 1e-12) tot += px; if (x >= a) one += px; }
  return { twoSided: r2(tot * 1000) / 1000, oneSidedGE: r2(one * 1000) / 1000 };
}
