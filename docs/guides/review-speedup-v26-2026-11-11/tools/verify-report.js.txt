function readTrace(dir) {
  const spans = new Map();
  for (const l of fs.readFileSync(path.join(dir, 'trace.jsonl'), 'utf8').split('\n')) {
    if (!l.trim()) continue;
    const e = JSON.parse(l);
    if (e.type === 'span_start') spans.set(e.span_id, { name: e.name, start: Date.parse(e.start_ts), attrs: e.attrs || {} });
    else if (e.type === 'span_end') { const s = spans.get(e.span_id); if (s) { s.end = Date.parse(e.end_ts); s.status = e.status; const lm = e.attrs && e.attrs.last_message_ts; s.iend = lm ? Date.parse(lm) : s.end; } }
  }
  return [...spans.values()];
}
function rowOf(id) { return lines.find((l) => l.startsWith('|') && l.includes(`\`${id}\``) && /\|\s*\d+\s*\|/.test(l)); }
function rowsOf(id) { return lines.filter((l) => l.startsWith('|') && l.includes(`\`${id}\``)); }
let bad = 0; const out = [];
function check(label, value, where) {
  const text = where === undefined ? REPORT : where;
  const ok = text.includes(String(value));
  if (!ok) bad += 1;
  out.push(`${ok ? 'OK  ' : 'FAIL'} ${label}: ${value}`);
}

  const stopped = ver.filter((s) => s.status === 'expired');
