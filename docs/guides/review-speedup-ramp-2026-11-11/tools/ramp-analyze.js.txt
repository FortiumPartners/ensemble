'use strict';
// Per-run analysis of the verify-concurrency ramp (kata j1wb). Scratch; a copy goes into the evidence directory as text.
// Inputs per run directory: trace.jsonl (the timing source), summary.json, timing.json, jobs.json, completion.json, concurrency.json,
// cap-proof.json, recall-guard.json, and the shared load.log. Nothing here opens an agent session or transcript.
//
// Definitions (the plan's S01 script, plan2/01_verifiers.py, and the earlier reports):
//   verifier duration  = dispatch (span start) -> result arrival (`last_message_ts` of the span end); a verifier stopped at its deadline has
//                        no arrival and is censored at (collected - dispatch), flagged expired
//   verify stage       = first verifier dispatch -> last verifier finish (the critical path's verifier step of `pr-review-cli timing`)
//   replay(durs, cap)  = S01's replay: start the verifiers in dispatch order, a slot starts the next as soon as one frees (heap), measured
//                        durations unchanged
const fs = require('fs');
const path = require('path');

const ts = (s) => Date.parse(s);
const mean = (xs) => (xs.length ? xs.reduce((a, b) => a + b, 0) / xs.length : null);
const pct = (xs, q) => { const s = [...xs].sort((a, b) => a - b); if (!s.length) return null; return s[Math.max(1, Math.ceil((q / 100) * s.length)) - 1]; };
const median = (xs) => { const s = [...xs].sort((a, b) => a - b); if (!s.length) return null; const m = s.length >> 1; return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2; };
const r2 = (x) => (x === null || x === undefined ? null : Math.round(x * 100) / 100);

function readTrace(file) {
  const spans = new Map();
  const order = [];
  for (const line of fs.readFileSync(file, 'utf8').split('\n')) {
    if (!line.trim()) continue;
    const e = JSON.parse(line);
    if (e.type === 'span_start') { spans.set(e.span_id, { name: e.name, start: ts(e.start_ts), attrs: e.attrs || {} }); order.push(e.span_id); }
    else if (e.type === 'span_end') { const s = spans.get(e.span_id); if (s) { s.end = ts(e.end_ts); s.status = e.status; s.endAttrs = e.attrs || {}; } }
  }
  return order.map((id) => spans.get(id));
}

/** S01's replay: `durs` (seconds) in dispatch order through `cap` slots; a freed slot starts the next verifier at once. */
function replay(durs, cap) {
  const n = durs.length;
  if (!n) return 0;
  if (cap >= n) return Math.max(...durs);
  const heap = durs.slice(0, cap).map((d) => d).sort((a, b) => a - b); // finish times of the first `cap`
  for (let i = cap; i < n; i += 1) {
    const t = heap.shift();
    const f = t + durs[i];
    let lo = 0; let hi = heap.length;
    while (lo < hi) { const mid = (lo + hi) >> 1; if (heap[mid] < f) lo = mid + 1; else hi = mid; }
    heap.splice(lo, 0, f);
  }
  return Math.max(...heap);
}

function loadWindow(loadLog, fromMs, toMs) {
  if (!fs.existsSync(loadLog)) return null;
  const rows = [];
  for (const line of fs.readFileSync(loadLog, 'utf8').split('\n')) {
    const m = /^(\S+Z) load1=([\d.]+) load5=([\d.]+) load15=([\d.]+) nodeProcs=(\d+) jestProcs=(\d+) fullsuiteLock=(\d)/.exec(line);
    if (!m) continue;
    const t = ts(m[1]);
    if (t >= fromMs && t <= toMs) rows.push({ t, load1: Number(m[2]), load5: Number(m[3]), load15: Number(m[4]), node: Number(m[5]), jest: Number(m[6]), lock: Number(m[7]) });
  }
  if (!rows.length) return { samples: 0 };
  const l1 = rows.map((r) => r.load1);
  return { samples: rows.length, load1Mean: r2(mean(l1)), load1Max: Math.max(...l1), load1Min: Math.min(...l1), samplesAbove20: l1.filter((x) => x > 20).length, load5Mean: r2(mean(rows.map((r) => r.load5))), jestProcsMax: Math.max(...rows.map((r) => r.jest)), samplesWithJest: rows.filter((r) => r.jest > 0).length, samplesWithFullSuiteLock: rows.filter((r) => r.lock === 1).length };
}

function analyzeRun(dir, { loadLog, caps = [8, 12, 17, 24] } = {}) {
  const spans = readTrace(path.join(dir, 'trace.jsonl'));
  const root = spans.find((s) => s.name === 'review');
  const t0 = root.start;
  const tEnd = root.end;
  const agents = spans.filter((s) => ['review.reviewer', 'review.security-lane', 'review.verifier'].includes(s.name)).map((s) => {
    const arrived = s.endAttrs.last_message_ts !== undefined;
    const finish = arrived ? ts(s.endAttrs.last_message_ts) : null;
    return { name: s.attrs.agent, role: s.attrs.role, wave: s.attrs.wave, refill: s.attrs.refill === true, cap: s.attrs.cap, running: s.attrs.running, start: s.start, collected: s.end, state: s.endAttrs.state, arrived, finish, endStatus: s.status, dur: arrived ? (finish - s.start) / 1000 : null, censored: arrived ? null : (s.end - s.start) / 1000 };
  });
  const reviewAgents = agents.filter((a) => a.role !== 'verifier');
  const verifiers = agents.filter((a) => a.role === 'verifier').sort((a, b) => (a.start - b.start) || (Number(a.name.split('-')[1]) - Number(b.name.split('-')[1])));
  const arrivedV = verifiers.filter((v) => v.arrived);
  const expiredV = verifiers.filter((v) => !v.arrived);
  const durOf = (v) => (v.arrived ? v.dur : v.censored);
  const vDurs = verifiers.map(durOf);
  const firstV = verifiers.length ? Math.min(...verifiers.map((v) => v.start)) : null;
  const lastVFinish = verifiers.length ? Math.max(...verifiers.map((v) => (v.arrived ? v.finish : v.collected))) : null;
  const lastReviewFinish = reviewAgents.length ? Math.max(...reviewAgents.map((a) => (a.arrived ? a.finish : a.collected))) : null;
  const reviewers = reviewAgents.filter((a) => a.role === 'reviewer');
  const slowestReviewer = reviewers.filter((a) => a.arrived).sort((a, b) => b.dur - a.dur)[0] || null;
  const lane = reviewAgents.find((a) => a.role === 'security') || null;
  // peak overlap of agents (dispatch -> finish or collection), over the whole run and over verifiers alone
  const overlap = (list) => {
    const ev = [];
    for (const a of list) { ev.push([a.start, 1]); ev.push([a.arrived ? a.finish : a.collected, -1]); }
    ev.sort((x, y) => x[0] - y[0] || x[1] - y[1]);
    let cur = 0; let peak = 0;
    for (const [, d] of ev) { cur += d; if (cur > peak) peak = cur; }
    return peak;
  };
  const arrivedDurs = arrivedV.map((v) => v.dur);
  const rec = {
    dir: path.basename(dir),
    wallMin: r2((tEnd - t0) / 60000),
    window: { startMs: t0, endMs: tEnd, startIso: new Date(t0).toISOString(), endIso: new Date(tEnd).toISOString() },
    rootMaxConcurrentAgents: root.endAttrs.max_concurrent_agents,
    reviewers: { count: reviewers.length, slowestName: slowestReviewer && slowestReviewer.name, slowestMin: slowestReviewer && r2(slowestReviewer.dur / 60), laneMin: lane && lane.dur !== null ? r2(lane.dur / 60) : null, expired: reviewAgents.filter((a) => !a.arrived).map((a) => a.name) },
    firstDispatchToReviewEndMin: r2((lastReviewFinish - Math.min(...reviewAgents.map((a) => a.start))) / 60000),
    handoffToVerifySec: firstV !== null ? r2((firstV - lastReviewFinish) / 1000) : null,
    verify: verifiers.length === 0 ? null : {
      count: verifiers.length, arrived: arrivedV.length, expired: expiredV.length, expiredNames: expiredV.map((v) => v.name),
      waves: [...new Set(verifiers.map((v) => v.wave))].length, refills: [...new Set(verifiers.filter((v) => v.refill).map((v) => v.wave))].length,
      firstWaveSize: verifiers.filter((v) => v.wave === 1).length,
      stageMin: r2((lastVFinish - firstV) / 60000), stageSec: r2((lastVFinish - firstV) / 1000),
      startedAtIso: new Date(firstV).toISOString(), startedOffsetMin: r2((firstV - t0) / 60000),
      slowestMin: r2(Math.max(...vDurs) / 60), slowestArrivedMin: arrivedDurs.length ? r2(Math.max(...arrivedDurs) / 60) : null,
      medianMin: r2(median(arrivedDurs) / 60), meanMin: r2(mean(arrivedDurs) / 60), p75Min: r2(pct(arrivedDurs, 75) / 60), p90Min: r2(pct(arrivedDurs, 90) / 60), minMin: r2(Math.min(...arrivedDurs) / 60),
      sumAgentMin: r2(vDurs.reduce((a, b) => a + b, 0) / 60),
      peakVerifierOverlap: overlap(verifiers), peakAnyAgentOverlap: overlap(agents),
      queueWaitMedianMin: r2(median(verifiers.map((v) => (v.start - firstV) / 1000)) / 60), lastStartOffsetMin: r2((Math.max(...verifiers.map((v) => v.start)) - firstV) / 60000),
      durationsSec: vDurs.map((d) => r2(d)),
      replaySec: Object.fromEntries([8, ...caps.filter((c) => c !== 8)].map((c) => [c, r2(replay(vDurs, c))])),
    },
  };
  if (rec.verify) {
    rec.verify.replayMin = Object.fromEntries(Object.entries(rec.verify.replaySec).map(([c, s]) => [c, r2(s / 60)]));
    rec.verify.stageVsOwnCapReplayMin = null; // filled by the caller, which knows the cap
  }
  // the gate and verification
  let sum = null; try { sum = JSON.parse(fs.readFileSync(path.join(dir, 'summary.json'), 'utf8')); } catch (e) { /* none */ }
  if (sum) {
    const g = sum.localCiGate || {};
    const gStart = g.startedAt ? ts(g.startedAt) : null;
    const gEnd = g.finishedAt ? ts(g.finishedAt) : null;
    rec.gate = { action: g.action, reason: g.reason, ok: g.ok, durationMin: g.durationMin, startedAt: g.startedAt, finishedAt: g.finishedAt, failures: g.failures, startedOffsetSec: gStart !== null ? r2((gStart - t0) / 1000) : null, finishedOffsetMin: gEnd !== null ? r2((gEnd - t0) / 60000) : null, verificationStartedAfterGateFinished: gEnd !== null && firstV !== null ? firstV >= gEnd : null, verifyStartMinusGateEndMin: gEnd !== null && firstV !== null ? r2((firstV - gEnd) / 60000) : null };
    rec.ledger = { candidates: sum.ledger.total, bySeverity: sum.ledger.bySeverity, present: sum.ledger.present, presentOnMajority: sum.ledger.presentOnMajority, unverified: sum.ledger.unverified, other: sum.ledger.other };
    rec.flags = { rereadResults: sum.flags.rereadResults.length, restated: sum.flags.restated.length, changedResults: sum.flags.changedResults.length, warnings: sum.flags.warnings.length };
    rec.stoppedAtDeadline = sum.stoppedAtDeadline; rec.incompleteVerifiers = sum.incompleteVerifiers;
    rec.orchestratorOnlySec = sum.timing.orchestratorOnly && sum.timing.orchestratorOnly.ms / 1000; rec.beforeFirstDispatchSec = sum.timing.beforeFirstDispatch / 1000;
    rec.criticalPath = sum.timing.criticalPath && { ms: sum.timing.criticalPath.ms, steps: (sum.timing.criticalPath.rounds || []).flatMap((r) => (r.steps || []).map((x) => ({ stage: x.stage, ms: x.ms, name: x.name, waves: x.waves }))) };
    rec.metricsAgents = sum.metrics && sum.metrics.agentsByRole;
  }
  // the harness job status (never a session file): statuses, model, thinking level, requests, cost
  let jobs = null; try { jobs = JSON.parse(fs.readFileSync(path.join(dir, 'jobs.json'), 'utf8')); } catch (e) { /* none */ }
  if (jobs) {
    const list = Object.entries(jobs).map(([name, j]) => ({ name, role: j.role, round: j.round, final: j.final, last: j.lastRunning, states: j.statesSeen }));
    const byRole = {};
    for (const j of list) {
      const b = (byRole[j.role] = byRole[j.role] || { n: 0, requests: 0, cost: 0, ctx: [], harnessDurSec: [], statuses: {}, notes: 0, models: {} });
      b.n += 1; if (j.last) { b.requests += j.last.requests || 0; b.cost += j.last.cost || 0; if (j.last.contextTokens) b.ctx.push(j.last.contextTokens); }
      if (j.final) { b.statuses[j.final.status + (j.final.killedAtDeadline ? '(killed)' : '')] = (b.statuses[j.final.status + (j.final.killedAtDeadline ? '(killed)' : '')] || 0) + 1; if (j.final.envelopeNote) b.notes += 1; if (j.final.durationMs) b.harnessDurSec.push(j.final.durationMs / 1000); const mk = (j.final.model || '?') + '|' + (j.final.thinking || '?'); b.models[mk] = (b.models[mk] || 0) + 1; }
    }
    rec.jobs = Object.fromEntries(Object.entries(byRole).map(([role, b]) => [role, { n: b.n, requestsTotal: b.requests, requestsMean: r2(b.requests / b.n), costTotalUsd: r2(b.cost), contextTokensMax: b.ctx.length ? Math.max(...b.ctx) : null, harnessDurMedianSec: r2(median(b.harnessDurSec)), harnessDurMaxSec: b.harnessDurSec.length ? r2(Math.max(...b.harnessDurSec)) : null, statuses: b.statuses, envelopeNotes: b.notes, modelsAndThinking: b.models }]));
    rec.harness = { agentsWithEnvelopeNote: list.filter((j) => j.final && j.final.envelopeNote).map((j) => j.name + ':' + j.final.envelopeNote), nonCompleted: list.filter((j) => j.final && j.final.status !== 'completed' && !j.final.killedAtDeadline).map((j) => j.name + ':' + j.final.status), killedAtDeadline: list.filter((j) => j.final && j.final.killedAtDeadline).map((j) => j.name), unusualWidgetStates: list.filter((j) => j.states.some((s) => s !== 'fallback')).map((j) => j.name + ':' + j.states.join('/')) };
  }
  let conc = null; try { conc = JSON.parse(fs.readFileSync(path.join(dir, 'concurrency.json'), 'utf8')); } catch (e) { /* none */ }
  if (conc) rec.driverPeakTrackedActive = conc.peakTrackedActive;
  let guard = null; try { guard = JSON.parse(fs.readFileSync(path.join(dir, 'recall-guard.json'), 'utf8')); } catch (e) { /* none */ }
  if (guard && guard.items) rec.recallGuard = { falseNegativeRate: guard.falseNegativeRate, notRaisedRate: guard.notRaisedRate, found: guard.items.filter((i) => i.status === 'found').map((i) => i.id), downgraded: guard.items.filter((i) => i.status === 'downgraded').map((i) => i.id), notRaised: guard.items.filter((i) => i.status === 'not-raised').map((i) => i.id), unmatchedImportant: (guard.unmatchedImportant || []).map((x) => x.id || x) };
  if (loadLog) {
    rec.load = { run: loadWindow(loadLog, t0, tEnd), verifyStage: firstV !== null ? loadWindow(loadLog, firstV, lastVFinish) : null, beforeRun15min: loadWindow(loadLog, t0 - 15 * 60000, t0) };
  }
  return rec;
}

module.exports = { analyzeRun, replay, readTrace, loadWindow, median, mean, pct, r2 };
