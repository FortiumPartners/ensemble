#!/usr/bin/env node
'use strict';
// Scratch, NOT committed (kata j1wb, plan v2 V2-2, measurement of the verify-concurrency ramp).
//
// `review-run-cli prepare`, run in-process by the real node, with exactly ONE seam. PR 207's base (feature/e2e-evidence-screenshots at
// cca69fe) has NO REVIEW.md, so no edit of REVIEW.md on dev can ever change #207's policy, and the runner takes the policy only from the
// base commit (`readBaseText` -> `loadPolicy({text, baseSha, sourceSha: baseSha})`). The two git reads of that file at the base are answered
// here from a scratch REVIEW.md whose policy block sets ONLY `maxConcurrentAgents`. Everything else is the committed code, unmodified:
// `readBaseText`, `loadPolicy`, `prepareRun`, the run state. No tracked REVIEW.md is edited anywhere.
//
// The seam is the documented injection point: `execute(command, { readJson, prepareReview })` of review-run-cli.js and
// `prepareReview(options, { git })` of pr-review-prepare.js ("`{ git, run }` replacements (see defaultDeps)").
//
// env: RAMP_REPO_ROOT (the worktree holding the committed code), RAMP_CAP (1..32), RAMP_SEAM_OUT (a file for the record of what was served),
//      RAMP_NO_SEAM=1 (control for the probe: the committed CLI path unchanged, same process shape)
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');

const ROOT = process.env.RAMP_REPO_ROOT;
const CAP = Number(process.env.RAMP_CAP);
const NO_SEAM = process.env.RAMP_NO_SEAM === '1';
if (!ROOT || !path.isAbsolute(ROOT) || (!NO_SEAM && !(Number.isInteger(CAP) && CAP >= 1 && CAP <= 32))) {
  process.stderr.write('RAMP_REPO_ROOT (absolute) and RAMP_CAP (integer 1..32) are required\n');
  process.exit(2);
}
const lib = path.join(ROOT, 'packages', 'development', 'lib');
const { main } = require(path.join(lib, 'review-run-cli.js'));
const { prepareReview, defaultDeps } = require(path.join(lib, 'pr-review-prepare.js'));

const TEXT = [
  '# REVIEW.md: scratch policy of the verify-concurrency ramp (kata j1wb)',
  '',
  'Served to `prepare` in place of the base commit\'s REVIEW.md. It sets one key.',
  '',
  '```ensemble-review-policy',
  JSON.stringify({ maxConcurrentAgents: CAP }, null, 2),
  '```',
  '',
].join('\n');
const BLOB = crypto.createHash('sha1').update(`blob ${Buffer.byteLength(TEXT)}\0`).update(TEXT).digest('hex');

const realGit = defaultDeps().git;
const served = [];
function git(args, opts) {
  // 1. the listing of REVIEW.md in the base tree (`readBaseText`: git ls-tree -z <baseSha> -- REVIEW.md)
  if (args[0] === 'ls-tree' && args.includes('-z') && args[args.length - 2] === '--' && args[args.length - 1] === 'REVIEW.md') {
    served.push({ call: 'ls-tree', baseSha: args[args.length - 3] });
    return { status: 0, stdout: `100644 blob ${BLOB}\tREVIEW.md\0`, stderr: '' };
  }
  // 2. the blob it names
  if (args[0] === 'cat-file' && args[1] === 'blob' && args[2] === BLOB) {
    served.push({ call: 'cat-file', blob: BLOB });
    const bytes = Buffer.from(TEXT, 'utf8');
    return { status: 0, stdout: opts && opts.buffer ? bytes : TEXT, stderr: '' };
  }
  return realGit(args, opts);
}

let payload;
try {
  payload = JSON.parse(fs.readFileSync(0, 'utf8'));
} catch (error) {
  process.stdout.write(`${JSON.stringify({ ok: false, error: `cannot read the payload: ${error.message}` })}\n`);
  process.exit(1);
}

const injected = NO_SEAM
  ? { readJson: () => payload }
  : { readJson: () => payload, prepareReview: (options) => prepareReview(options, { git }) };

main(['prepare'], injected).then((code) => {
  if (process.env.RAMP_SEAM_OUT) {
    fs.writeFileSync(process.env.RAMP_SEAM_OUT, `${JSON.stringify({
      seam: !NO_SEAM, cap: NO_SEAM ? null : CAP, scratchReviewMd: NO_SEAM ? null : TEXT,
      scratchReviewMdBlob: NO_SEAM ? null : BLOB, scratchReviewMdSha256: NO_SEAM ? null : crypto.createHash('sha256').update(TEXT).digest('hex'), served,
    }, null, 1)}\n`);
  }
  process.exit(code);
});
